export function registerConfigResource(server: McpServer) {
  server.resource(
    "config",
    "mcp://config",
    async (uri) => ({
      contents: [{
        uri: uri.href,
        text: JSON.stringify({ version: "1.0.0" }, null, 2),
      }],
    })
  );
}
