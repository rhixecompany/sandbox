## Swift MCP Server Pattern

```swift
// Sources/MyMCPServer/main.swift
import MCP
import ServiceLifecycle
import Logging

let logger = Logger(label: "mcp.server")

let server = try await MCPServer(
    name: "swift-mcp-server",
    version: "1.0.0",
    tools: [
        // Tool definitions
    ],
    resources: [
        // Resource definitions
    ]
)

try await withThrowingTaskGroup(of: Void.self) { group in
    group.addTask {
        try await ServiceLifecycle.run(server)
    }
    // Handle shutdown signals
    for await _ in SignalHandler.signals() {
        group.cancelAll()
    }
}
```
