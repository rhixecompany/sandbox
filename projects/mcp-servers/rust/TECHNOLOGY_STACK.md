## Entry Point (`src/main.rs`)

```rust
use rmcp::{
    ServerHandler, 
    model::*, 
    service::RequestContext,
    tool, 
    transport::stdio::stdio_server,
};
use std::sync::Arc;

#[derive(Clone)]
struct MyMcpServer;

#[tool(tool_box)]
impl MyMcpServer {
    #[tool(description = "Greet a person by name")]
    async fn greet(&self, name: String) -> Result<String, anyhow::Error> {
        Ok(format!("Hello, {}!", name))
    }
}

#[tool::tool_box]
impl MyMcpServer {
    // Additional tools can be added here
}

#[tokio::main]
async fn main() -> Result<(), anyhow::Error> {
    let server = MyMcpServer;
    let service = server.serve(stdio_server()).await?;
    service.waiting().await?;
    Ok(())
}
```

### Manual implementation (if needed)

```rust
// Alternative manual tool definition
use rmcp::{model::*, ServerHandler, RequestContext};

impl ServerHandler for MyMcpServer {
    fn list_tools(&self) -> Vec<Tool> {
        vec![Tool::new(
            "greet",
            "Greet a person by name",
            ToolInputSchema::object()
                .property("name", Schema::string("The name of the person to greet"))
                .required(vec!["name"])
        )]
    }
    
    async fn call_tool(
        &self, 
        name: &str, 
        arguments: serde_json::Value, 
        _context: RequestContext
    ) -> Result<CallToolResult, anyhow::Error> {
        match name {
            "greet" => {
                let name = arguments.get("name")
                    .and_then(|v| v.as_str())
                    .unwrap_or("World");
                Ok(CallToolResult::text(format!("Hello, {}!", name)))
            }
            _ => Err(anyhow::anyhow!("Unknown tool: {}", name)),
        }
    }
}
```
