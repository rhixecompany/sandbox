<?php

require_once __DIR__ . '/vendor/autoload.php';

use McpServer\Server;
use McpServer\Tools\GreetTool;
use Loganmarchione\Mcp\Server\StdioServerTransport;

$server = new Server("php-mcp-server", "1.0.0");

// Run with STDIO transport

## Tool Implementation (`src/McpServer/Tools/GreetTool.php`)

```php
<?php

namespace McpServer\Tools;

use Loganmarchione\Mcp\Server\Tool\ToolInterface;
use Loganmarchione\Mcp\Schema\Tool;
use Loganmarchione\Mcp\Schema\ToolInputSchema;

class GreetTool implements ToolInterface
{
    public function getDefinition(): Tool
    {
        return new Tool(
            name: 'greet',
            description: 'Greet a person by name',
            inputSchema: new ToolInputSchema(
                type: 'object',
                properties: [
                    'name' => [
                        'type' => 'string',
                        'description' => 'The name of the person to greet',
                    ],
                ],
                required: ['name'],
            ),
        );
    }

    public function execute(array $arguments): array
    {
        $name = $arguments['name'] ?? 'World';
        
        return [
            'content' => [
                [
                    'type' => 'text',
                    'text' => "Hello, {$name}!",
                ],
            ],
        ];
    }
}
```
