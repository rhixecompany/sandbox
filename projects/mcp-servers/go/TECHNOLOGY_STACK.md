## Tool Implementation (`tools/greet.go`)

```go
package tools

import (
    "context"
    
    "github.com/modelcontextprotocol/go-sdk/mcp"
)

type GreetTool struct{}

func NewGreetTool() *GreetTool {
    return &GreetTool{}
}

func (t *GreetTool) Name() string {
    return "greet"
}

func (t *GreetTool) Description() string {
    return "Greet a person by name"
}

func (t *GreetTool) InputSchema() *mcp.ToolInputSchema {
    return &mcp.ToolInputSchema{
        Type: "object",
        Properties: map[string]*mcp.ToolInputSchemaProperty{
            "name": {
                Type:        "string",
                Description: "The name of the person to greet",
            },
        },
        Required: []string{"name"},
    }
}

func (t *GreetTool) Handle(ctx context.Context, req *mcp.CallToolRequest) (*mcp.CallToolResult, error) {
    name, ok := req.Params.Arguments["name"].(string)
    if !ok || name == "" {
        return nil, mcp.NewInvalidParamsError("Missing 'name' parameter")
    }
    
    return &mcp.CallToolResult{
        Content: []mcp.Content{
            mcp.NewTextContent("Hello, " + name + "!"),
        },
    }, nil
}
```
